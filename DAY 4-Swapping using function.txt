#include<stdio.h>
int swap(int *a,int *b)
{
    int c;
    c=*a;
    *a=*b;
    *b=c;
}
int main()
{
    int x,y;
    printf("Enter x:");
    scanf("%d",&x);
    printf("Enter y:");
    scanf("%d",&y);
    printf("before swap x is %d and y is %d\n",x,y);
    swap(&x,&y);
    printf("After swap x is %d and y is %d\n",x,y);
    printf("x=%d\ny=%d",x,y);
    return 0;
}
